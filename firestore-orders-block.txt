// ComfortFeet: orders (cash on delivery). Anyone can create a well-formed order,
// only the admin email can read it and change its status, nobody can delete.
function isShopAdmin() {
  return request.auth != null && request.auth.token.email == 'mrjomaiomar@gmail.com';
}
function validOrder(d) {
  return d.keys().hasOnly(['customer', 'items', 'subtotal', 'delivery', 'total', 'payment', 'status', 'createdAt'])
    && d.payment == 'cash_on_delivery'
    && d.status == 'nouvelle'
    && d.createdAt == request.time
    && d.items is list && d.items.size() > 0 && d.items.size() <= 30
    && d.total is number && d.total > 0 && d.total < 5000
    && d.customer.fullName is string && d.customer.fullName.size() >= 3 && d.customer.fullName.size() <= 80
    && d.customer.phone is string && d.customer.phone.matches('^[2-9][0-9]{7}$')
    && d.customer.address is string && d.customer.address.size() <= 200
    && d.customer.note is string && d.customer.note.size() <= 300;
}
match /orders/{id} {
  allow create: if validOrder(request.resource.data);
  allow read: if isShopAdmin();
  allow update: if isShopAdmin()
    && request.resource.data.status in ['nouvelle', 'confirmée', 'expédiée', 'livrée', 'annulée']
    && request.resource.data.customer == resource.data.customer
    && request.resource.data.items == resource.data.items
    && request.resource.data.total == resource.data.total
    && request.resource.data.createdAt == resource.data.createdAt;
  allow delete: if false;
}
